<script>
  var DATA = [], STATUSES = [], SOURCES = [], FILTER = 'Все';

  function load(){
    google.script.run.withSuccessHandler(function(res){
      DATA = res.rows; STATUSES = res.statuses; SOURCES = res.sources || []; render();
    }).withFailureHandler(function(e){
      document.getElementById('tb').innerHTML =
        '<tr><td colspan="7" class="empty">Не удалось загрузить: ' + e.message + '</td></tr>';
    }).apiList();
  }

  function setFilter(f, el){
    FILTER = f;
    [].forEach.call(document.querySelectorAll('.chip'), function(c){ c.classList.toggle('on', c === el); });
    render();
  }

  function esc(s){ return String(s||'').replace(/[&<>"]/g, function(c){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]; }); }

  function digits(p){ return String(p||'').replace(/\D/g,''); }

  function visible(){
    var q = (document.getElementById('q').value || '').toLowerCase().trim();
    return DATA.filter(function(r){
      if (FILTER !== 'Все' && r.status !== FILTER) return false;
      if (!q) return true;
      return [r.name,r.phone,r.email,r.objectType,r.comment,r.intent,r.owner,r.notes]
        .join(' ').toLowerCase().indexOf(q) >= 0;
    });
  }

  function render(){
    var rows = visible();
    var today = new Date(); today.setHours(0,0,0,0);
    var k = {
      all: DATA.length,
      today: DATA.filter(function(r){ return r.ts >= today.getTime(); }).length,
      nw: DATA.filter(function(r){ return r.status === 'Новая'; }).length,
      work: DATA.filter(function(r){ return r.status === 'В работе' || r.status === 'КП отправлено'; }).length,
      deal: DATA.filter(function(r){ return r.status === 'Договор'; }).length
    };
    var won = DATA.filter(function(r){ return r.status === 'Договор'; })
                  .reduce(function(a,r){ return a + (Number(r.amount)||0); }, 0);
    var pipe = DATA.filter(function(r){ return r.status === 'В работе' || r.status === 'КП отправлено'; })
                   .reduce(function(a,r){ return a + (Number(r.amount)||0); }, 0);
    document.getElementById('kpis').innerHTML =
      kpi(k.nw,'новых заявок','hot') + kpi(k.today,'сегодня') + kpi(k.work,'в работе') +
      kpi(k.deal,'договоров') + kpi(money(won),'сумма договоров','hot') + kpi(money(pipe),'в работе, ₽');

    if (!rows.length){
      document.getElementById('tb').innerHTML = '<tr><td colspan="7" class="empty">Заявок пока нет</td></tr>';
      return;
    }

    document.getElementById('tb').innerHTML = rows.map(function(r){
      var si = STATUSES.indexOf(r.status); if (si < 0) si = 0;
      var tel = digits(r.phone);
      var req = [r.objectType, r.area ? r.area + ' м²' : ''].filter(Boolean).join(' · ');
      return '<tr>' +
        '<td data-l="Заявка"><div class="nm">' + esc(r.id) + '</div><div class="sub">' + esc(r.date) + '</div>' +
          '<div class="sub">' + esc(r.intent) + '</div></td>' +
        '<td data-l="Контакт"><div class="nm">' + esc(r.name || '—') + '</div>' +
          (r.phone ? '<div class="sub"><a href="tel:+' + tel + '">' + esc(r.phone) + '</a></div>' : '') +
          (r.email ? '<div class="sub"><a href="mailto:' + esc(r.email) + '">' + esc(r.email) + '</a></div>' : '') +
          (tel ? '<div class="msg"><a href="https://wa.me/' + tel + '" target="_blank" title="WhatsApp">💬</a>' +
                 '<a href="https://t.me/+' + tel + '" target="_blank" title="Telegram">✈️</a></div>' : '') +
        '</td>' +
        '<td data-l="Статус"><span class="st s' + si + '">' + esc(r.status) + '</span>' +
          '<select style="margin-top:7px" onchange="save(' + r.line + ',\'status\',this.value,this)">' +
          STATUSES.map(function(s){
            return '<option' + (s === r.status ? ' selected' : '') + '>' + esc(s) + '</option>';
          }).join('') + '</select></td>' +
        '<td data-l="Запрос">' + (req ? '<div class="nm">' + esc(req) + '</div>' : '') +
          (r.comment ? '<div class="sub cmt">' + esc(r.comment) + '</div>' : '') +
          (r.file ? '<div class="sub">📎 ' + esc(r.file) + '</div>' : '') +
          (r.page ? '<div class="sub"><a href="' + esc(r.page) + '" target="_blank">страница</a></div>' : '') +
        '</td>' +
        '<td data-l="Сумма договора"><div class="money">' + (r.amount ? money(r.amount) : '—') + '</div>' +
          '<input class="inp sum" style="margin-top:6px" value="' + (r.amount || '') + '" placeholder="сумма"' +
          ' onchange="save(' + r.line + ',\'amount\',this.value,this)"></td>' +
        '<td data-l="Ответственный"><input class="inp" value="' + esc(r.owner) + '" placeholder="кто ведёт"' +
          ' onchange="save(' + r.line + ',\'owner\',this.value,this)"></td>' +
        '<td data-l="Заметки"><textarea placeholder="итог звонка, договорённости…"' +
          ' onchange="save(' + r.line + ',\'notes\',this.value,this)">' + esc(r.notes) + '</textarea>' +
          '<div class="saved" id="sv' + r.line + '">сохранено</div></td>' +
      '</tr>';
    }).join('');
  }

  function money(v){
    var n = Number(v) || 0;
    return n.toLocaleString('ru-RU') + ' ₽';
  }

  function kpi(v,label,cls){
    return '<div class="kpi ' + (cls||'') + '"><b>' + v + '</b><span>' + label + '</span></div>';
  }

  function save(line, field, value, el){
    var patch = {}; patch[field] = value;
    el.style.borderColor = '#c08f2a';
    google.script.run.withSuccessHandler(function(){
      el.style.borderColor = '';
      var s = document.getElementById('sv' + line);
      if (s){ s.classList.add('on'); setTimeout(function(){ s.classList.remove('on'); }, 1200); }
      var row = DATA.filter(function(r){ return r.line === line; })[0];
      if (row){
        row[field] = field === 'amount' ? (parseFloat(String(value).replace(/[^0-9.,-]/g,'').replace(',','.')) || '') : value;
        if (field === 'status' || field === 'amount') render();
      }
    }).apiUpdate(line, patch);
  }

  function exportCsv(){
    var rows = visible();
    var head = ['ID','Дата','Статус','Имя','Телефон','E-mail','Тип объекта','Площадь','Комментарий','Источник','Сумма','Ответственный','Заметки'];
    var body = rows.map(function(r){
      return [r.id,r.date,r.status,r.name,r.phone,r.email,r.objectType,r.area,r.comment,r.intent,r.amount,r.owner,r.notes]
        .map(function(v){ return '"' + String(v||'').replace(/"/g,'""') + '"'; }).join(';');
    });
    var csv = '﻿' + [head.join(';')].concat(body).join('\n');
    var a = document.createElement('a');
    a.href = 'data:text/csv;charset=utf-8,' + encodeURIComponent(csv);
    a.download = 'zayavki-lider.csv';
    a.click();
  }

  function openAdd(){
    var st = document.getElementById('f_status');
    st.innerHTML = STATUSES.map(function(s){ return '<option>' + esc(s) + '</option>'; }).join('');
    var sr = document.getElementById('f_source');
    sr.innerHTML = (SOURCES.length ? SOURCES : ['Звонок']).map(function(s){ return '<option>' + esc(s) + '</option>'; }).join('');
    ['f_name','f_phone','f_email','f_object','f_area','f_amount','f_owner','f_comment','f_notes']
      .forEach(function(id){ document.getElementById(id).value = ''; });
    document.getElementById('f_err').style.display = 'none';
    document.getElementById('ov').classList.add('on');
    document.getElementById('f_name').focus();
  }

  function closeAdd(){ document.getElementById('ov').classList.remove('on'); }

  function saveNew(){
    var v = function(id){ return document.getElementById(id).value.trim(); };
    var err = document.getElementById('f_err');
    if (!v('f_name') && !v('f_phone')){
      err.textContent = 'Укажите хотя бы имя или телефон.';
      err.style.display = 'block';
      return;
    }
    err.style.display = 'none';
    var btn = document.getElementById('f_save');
    btn.disabled = true; btn.textContent = 'Добавляем…';
    google.script.run
      .withSuccessHandler(function(){
        btn.disabled = false; btn.textContent = 'Добавить заявку';
        closeAdd(); load();
      })
      .withFailureHandler(function(e){
        btn.disabled = false; btn.textContent = 'Добавить заявку';
        err.textContent = 'Не удалось сохранить: ' + e.message;
        err.style.display = 'block';
      })
      .apiAdd({
        name: v('f_name'), phone: v('f_phone'), email: v('f_email'),
        objectType: v('f_object'), area: v('f_area'), amount: v('f_amount'),
        source: v('f_source'), status: v('f_status'), owner: v('f_owner'),
        comment: v('f_comment'), notes: v('f_notes')
      });
  }

  document.addEventListener('keydown', function(e){ if (e.key === 'Escape') closeAdd(); });

  load();
  setInterval(load, 60000);
</script>
